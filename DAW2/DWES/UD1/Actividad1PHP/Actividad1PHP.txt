function factorial($num) {
  $result = 1;
  for ($i = 1; $i <= $num; $i++){
  	$result *= $i;
  }
  return $result;
}

function calcularPromedio($arr){
	$suma = array_sum($arr);
    $cantidad = count($arr);
	
    if ($cantidad > 0){
    	return $suma / $cantidad;
    } else {
    	return 0;
    }
}

$matriz = array(
    array(1, 2, 3),
    array(4, 5, 6),
    array(7, 8, 9)
);

$suma = 0;

for ($i = 0; $i < count($matriz); $i++) { //filas
    for ($j = 0; $j < count($matriz[$i]); $j++) { //columnas
        $suma += $matriz[$i][$j]; // la suma de todos los elementos
    }
}

foreach ($matriz as $fila) { //enseñar la matriz
    echo "[" . implode(", ", $fila) . "]<br>";
}
