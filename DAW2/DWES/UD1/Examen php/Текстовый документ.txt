// Ejercicio 6: Función para contar vocales en una frase
function contarVocales($texto) {
    $vocales = "aeiou";
    $contador = 0;
    $texto = strtolower($texto);
    for ($i = 0; $i < strlen($texto); $i++) {
        if (strpos($vocales, $texto[$i]) !== false) {
            $contador++;
        }
    }
    return $contador;
}

// 🎮 Simulación: “Adivina el número secreto”
// PHP genera un número aleatorio entre 1 y 50
// El usuario tiene 5 intentos simulados (automáticos) para adivinar

echo "🎯 <strong>JUEGO: Adivina el número secreto</strong><br>";

$numeroSecreto = rand(1, 50);
$intentos = 5;
$ganador = false;

for ($i = 1; $i <= $intentos; $i++) {
    $intento = rand(1, 50); // simulación del intento del jugador
    echo "Intento $i: el jugador dice $intento... ";

    if ($intento == $numeroSecreto) {
        echo "✅ ¡Correcto! Has adivinado el número secreto ($numeroSecreto) en el intento $i.<br>";
        $ganador = true;
        break;
    } elseif ($intento < $numeroSecreto) {
        echo "El número secreto es mayor.<br>";
    } else {
        echo "El número secreto es menor.<br>";
    }

// Ejercicio 6: Función para invertir un texto manualmente
function invertirTexto($texto) {
    $invertido = "";
    for ($i = strlen($texto) - 1; $i >= 0; $i--) {
        $invertido .= $texto[$i];
    }
    return $invertido;
}
$texto = "PHP es divertido";
echo "6️⃣ Texto original: $texto<br>";
echo "Texto invertido: " . invertirTexto($texto) . "<br><br>";

// 🎯 Proyecto: “Simulación de lanzamiento de dados”

// Se lanzan 2 dados durante 10 rondas.
// Se muestra el resultado de cada ronda y se determina quién gana más.

echo "🎲 <strong>JUEGO: Lanzamiento de dados</strong><br>";

$resultados = [
    "Jugador 1" => 0,
    "Jugador 2" => 0,
    "Empates" => 0
];

$rondas = 10;

for ($r = 1; $r <= $rondas; $r++) {
    $dado1 = rand(1, 6);
    $dado2 = rand(1, 6);
    echo "Ronda $r: Jugador 1 = $dado1, Jugador 2 = $dado2 → ";

    if ($dado1 > $dado2) {
        echo "Gana Jugador 1<br>";
        $resultados["Jugador 1"]++;
    } elseif ($dado2 > $dado1) {
        echo "Gana Jugador 2<br>";
        $resultados["Jugador 2"]++;
    } else {
        echo "Empate<br>";
        $resultados["Empates"]++;
    }
}

echo "<br>Resultados finales:<br>";
echo "Jugador 1: {$resultados['Jugador 1']} rondas ganadas<br>";
echo "Jugador 2: {$resultados['Jugador 2']} rondas ganadas<br>";
echo "Empates: {$resultados['Empates']}<br>";

if ($resultados["Jugador 1"] > $resultados["Jugador 2"]) {
    echo "<br>🏆 El ganador final es el Jugador 1!<br>";
} elseif ($resultados["Jugador 2"] > $resultados["Jugador 1"]) {
    echo "<br>🏆 El ganador final es el Jugador 2!<br>";
} else {
    echo "<br>🤝 Empate general!<br>";
}
