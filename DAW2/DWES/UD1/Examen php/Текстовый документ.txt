// Ejercicio 6: Función para contar vocales en una frase
function contarVocales($texto) {
    $vocales = "aeiou";
    $contador = 0;
    $texto = strtolower($texto);
    for ($i = 0; $i < strlen($texto); $i++) {
        if (strpos($vocales, $texto[$i]) !== false) {
            $contador++;
        }
    }
    return $contador;
}

// Ejercicio 6: Función para invertir un texto manualmente
function invertirTexto($texto) {
    $invertido = "";
    for ($i = strlen($texto) - 1; $i >= 0; $i--) {
        $invertido .= $texto[$i];
    }
    return $invertido;
}
$texto = "PHP es divertido";
echo "6️⃣ Texto original: $texto<br>";
echo "Texto invertido: " . invertirTexto($texto) . "<br><br>";

// Ejercicio 6: Función que devuelve la palabra más corta de una frase
function palabraMasCorta($frase) {
    $palabras = explode(" ", $frase);
    $masCorta = $palabras[0];
    foreach ($palabras as $palabra) {
        if (strlen($palabra) < strlen($masCorta)) {
            $masCorta = $palabra;
        }
    }
    return $masCorta;
}

// 🚗 Proyecto: “Simulador de carreras”
//
// 3 coches compiten en 5 vueltas. Cada vuelta se genera un tiempo aleatorio.
// Gana el coche con menor tiempo total.

echo "🏁 <strong>SIMULADOR DE CARRERA DE COCHES</strong><br><br>";

$coches = [
    "Coche A" => [],
    "Coche B" => [],
    "Coche C" => []
];

$vueltas = 5;

// Generar tiempos aleatorios
foreach ($coches as $nombre => &$tiempos) {
    for ($i = 1; $i <= $vueltas; $i++) {
        $tiempos[$i] = rand(50, 120); // tiempo entre 50 y 120 segundos
    }
}
unset($tiempos); // buena práctica para evitar referencias activas

// Mostrar tiempos y calcular total
$resultados = [];
foreach ($coches as $nombre => $tiempos) {
    $total = array_sum($tiempos);
    $resultados[$nombre] = $total;
    echo "<strong>$nombre</strong>: ";
    echo implode("s, ", $tiempos) . "s → Total: {$total}s<br>";
}

// Determinar el ganador
$ganador = array_keys($resultados, min($resultados))[0];
echo "<br>🏆 El ganador de la carrera es: <strong>$ganador</strong> con " . min($resultados) . " segundos.<br>";

echo "<br><br>Fin del examen 🧾";
